\addchap{Notation}

The two literatures disagree on almost every normalization: the sign of the free energy, the scale of the couplings, the name of the aspect ratio, the factor in front of a Hamiltonian. The chapters share the conventions below. Where a chapter uses one of these letters for something else, it defines the local meaning where it introduces it, and the last paragraph of each chapter's notes, \emph{Translating the literature}, maps the notation of its main sources onto this one.

\addsec*{Sizes and limits}

\begin{tabular}{@{}p{0.27\textwidth}p{0.69\textwidth}@{}}
$N$ & the number of spins or variables in a spin glass or a matrix model; in \cref{ch:kernels}, the number of random features \\
$d$, $n$ & the dimension of the unknown in a learning problem, and the number of samples or patterns \\
$\alpha=n/d$ & the load or sample ratio (Gardner's $\alpha$; Donoho's $\delta$); $\gamma=d/n$ only when quoting sources that use it \\
$\psi=N/d$ & the number of random features per dimension (\cref{ch:kernels}) \\
proportional limit & $n,d\to\infty$ with $n/d\to\alpha$ fixed; the number of iterations or the time horizon is fixed unless stated
\end{tabular}

\noindent $\Prob$ and $\E$ without a subscript are over the disorder (the data); a subscript names other randomness. Every limit statement says in what sense it holds. Logarithms are natural.

\addsec*{Gaussian notation}

\begin{tabular}{@{}p{0.27\textwidth}p{0.69\textwidth}@{}}
$Z$, $z$ & a standard Gaussian variable, $\Normal(0,1)$ \\
$\int f(z)\Dz$ & the physicists' notation for $\E f(Z)$ \\
$\phi$, $\Phi$ & the standard normal density and distribution function \\
$H(x)$ & the Gardner function $\Prob(Z>x)=1-\Phi(x)$ \\
GOE$(N)$ & the symmetric Gaussian matrix with off-diagonal entries of variance $1/N$ and diagonal entries of variance $2/N$; its spectrum fills $[-2,2]$
\end{tabular}

\addsec*{Free entropy, Gibbs measures and overlaps}

\begin{tabular}{@{}p{0.27\textwidth}p{0.69\textwidth}@{}}
$Z_N$ & the partition function; in inference, the evidence (marginal likelihood) \\
$\phi_N=\frac1N\E\log Z_N$, $\phi$ & the free entropy and its limit (a mathematician's ``pressure''); the physicist's free energy is $f=-\phi/\beta$, quoted only where a chapter says so \\
$\gibbs{\cdot}$, $\E$ & the Gibbs (posterior) average, and the average over the disorder (the data) \\
$\sigma^1,\sigma^2,\dots$ & replicas: independent samples from the same Gibbs measure \\
$R_{12}=\frac1N\ip{\sigma^1}{\sigma^2}$ & the overlap of two replicas, a random variable \\
$q$ & the limit, or the typical value, of $R_{12}$ in a phase where it concentrates; a fixed point of the state equations is written with a star \\
$m=\frac1N\ip{\sigma}{x^*}$ & the overlap with the truth $x^*$ \\
$\mathrm{mmse}(\gamma)$ & the minimum mean squared error of the scalar Gaussian channel $Y=\sqrt\gamma\,X+Z$
\end{tabular}

\addsec*{Models}

\begin{tabular}{@{}p{0.27\textwidth}p{0.69\textwidth}@{}}
REM & $Z_N=\sum_{\sigma\in\{\pm1\}^N}e^{\beta\sqrt N X_\sigma}$, with $X_\sigma$ i.i.d.\ $\Normal(0,1)$ and $\beta_c=\sqrt{2\log2}$ \\
SK & $H_N(\sigma)=\frac{\beta}{\sqrt N}\sum_{i<j}g_{ij}\sigma_i\sigma_j+h\sum_i\sigma_i$, Gibbs weight $e^{H_N(\sigma)}$, covariance $N\xi(R_{\sigma\tau})+O(1)$ with $\xi(q)=\beta^2q^2/2$, and $\beta_c=1$ \\
spiked Wigner & $A=\frac\lambda N xx^\top+W$, with $W\sim\mathrm{GOE}(N)$ and $x_i$ i.i.d.\ from a prior $P_0$ with $\E x_i^2=1$; the BBP transition is at $\lambda=1$ \\
perceptron & patterns $g_\mu\sim\Normal(0,I_d)$, weights $w$ on the sphere $\norm w^2=d$ or in $\{\pm1\}^d$, constraints $\ip{g_\mu}{w}/\sqrt d\ge\kappa$ with margin $\kappa$, and Gardner's capacity $\alpha_c(\kappa)$ \\
GLM & $y_\mu=\varphi_{\mathrm{out}}(\ip{x_\mu}{w^*}/\sqrt d,A_\mu)$ with $x_\mu\sim\Normal(0,I_d)$ and $w^*$ i.i.d.\ from $P_0$; the thresholds $\alpha_{\mathrm{IT}}$ (information-theoretic) and $\alpha_{\mathrm{AMP}}$ (algorithmic) \\
regression & $x_i\sim\Normal(0,\Sigma)$ with $\Sigma=I_d$ unless stated, $y=\ip x{\beta^*}+\sigma\varepsilon$, $r^2=\ip{\beta^*}{\Sigma\beta^*}$; ridge minimizes $\frac1{2n}\norm{y-X\beta}^2+\frac\lambda2\norm\beta^2$; the excess risk is $R=\E_x(\ip x{\hat\beta}-\ip x{\beta^*})^2$ and the test error $R+\sigma^2$ \\
classification & labels $y\in\{\pm1\}$, test error $\Prob(y\ne\sgn\ip x{\hat\beta})$; $\kappa$ is always a margin \\
kernels & eigenvalues $\eta_k$ of the kernel operator, and the effective ridge $\lambda_{\mathrm{eff}}$ \\
spherical $p$-spin & $H_N$ on the sphere of radius $\sqrt N$ with $\E H_N(\sigma)H_N(\sigma')=N R(\sigma,\sigma')^p$, energies per spin, ground state $-E_0(p)$ and threshold $-E_\infty(p)$, $E_\infty=2\sqrt{(p-1)/p}$; physicists' energies are these divided by $\sqrt2$
\end{tabular}

\addsec*{The status words}

\begin{tabular}{@{}p{0.27\textwidth}p{0.69\textwidth}@{}}
\proved & a theorem in a refereed publication, a classical result, or a theorem proved completely in this book, with its source \\
\provedunder & proved under hypotheses beyond those of the prediction (a prior, a range of parameters, Gaussian data, smoothness, a fixed number of iterations), which the board names \\
\provedpreprint & proved only in an unrefereed preprint, named with its arXiv number and version \\
\conjectured & believed, with the evidence named (physics, numerics, partial results) \\
\heuristicstatus & an argument that claims no exactness \\
\knownfalse & a counterexample, or a regime where the prediction fails, named \\
\openq & no one knows, and no one has a strong conjecture; or a question posed in a cited source
\end{tabular}
